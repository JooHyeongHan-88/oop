class Account:

from functools import reduce

def deposit(balance, amount):
    return balance + amount   # 새 값 반환

amounts = [100, -20, 50]

balance = reduce(deposit,     # 2. 누적
    filter(lambda a: a > 0, amounts), 0)  # 1. 거르기
# filter → reduce 체이닝: 150

class Dog:                          # (1)

문제 01 · 흩어진 데이터

“이 dict에 어떤 키가 있었더라?”

✕ 절차 지향dict + 여기저기 흩어진 함수

user = {"name": "Kim", "email": "kim@a.com"}

# users.py
def change_email(user, new): ...
# mail.py
def send_mail(user, msg): ...
# auth.py
def validate_user(user): ...

# 키 이름을 기억해서 써야 함

@dataclass
class User:

문제 02 · 무방비한 상태

“잔액이 왜 마이너스지?”

✕ 절차 지향규칙은 함수에, 데이터는 누구나 수정

account = {"balance": 1000}

def withdraw(acc, amount):
    if amount > acc["balance"]:
        raise ValueError("잔액 부족")
    acc["balance"] -= amount


# 다른 파일 어딘가에서...
account["balance"] = -5000   # 규칙 우회!
✓ 객체 지향캡슐화 — 정해진 문으로만 출입

class Account:
    def __init__(self, balance=0):
        self._balance = balance

    @property
    def balance(self):          # 읽기 전용
        return self._balance

    def withdraw(self, amount):
        if amount > self._balance:
            raise ValueError("잔액 부족")
        self._balance -= amount

acc = Account(1000)
acc.balance = -5000   # AttributeError
문제

검증 규칙을 건너뛰는 길이 열려 있다. 버그가 나면 어디서 바꿨는지 전체를 뒤져야 한다.

개선

상태 변경은 메서드를 통해서만. 규칙이 항상 지켜지고, 변경 지점이 한 곳이다.

문제 03 · if / elif 지옥

“삼각형 추가해 주세요” — 고칠 곳이 몇 군데?

✕ 절차 지향타입마다 분기, 분기가 함수마다 반복

def area(s):
    if s["type"] == "circle":
        return 3.14 * s["r"] ** 2
    elif s["type"] == "rect":
        return s["w"] * s["h"]
    # 삼각형? → 여기 수정

def draw(s):
    if s["type"] == "circle": ...
    elif s["type"] == "rect": ...
    # → 여기도 수정
# save(), to_json() ... 전부 수정
✓ 객체 지향다형성 — 각자 자기 일을 안다

class Circle:
    def __init__(self, r): self.r = r
    def area(self): return 3.14 * self.r ** 2

class Rect:
    def __init__(self, w, h): self.w, self.h = w, h
    def area(self): return self.w * self.h

class Triangle:          # 추가 = 클래스 하나
    def __init__(self, b, h): self.b, self.h = b, h
    def area(self): return self.b * self.h / 2

for s in [Circle(1), Rect(2, 3), Triangle(4, 5)]:
    print(s.area())      # 타입을 몰라도 OK
문제

새 종류 하나에 모든 분기를 찾아 고쳐야 한다. 하나라도 놓치면 버그.

개선

새 종류 = 새 클래스 추가. 기존 코드는 건드리지 않는다.

문제 04 · 복붙 중복

“로그 형식 바꿔 주세요” — 복붙한 곳이 몇 군데?

✕ 절차 지향조금씩 다른 코드를 복사해서 수정

def save_user(u):
    log("saving...")
    check_email(u)
    db.insert("users", u)
    log("done")

def save_order(o):
    log("saving...")       # 복붙
    check_qty(o)           # 여기만 다름
    db.insert("orders", o)
    log("done")            # 복붙

# save_product, save_review ...
✓ 객체 지향공통 흐름은 부모에, 차이만 자식에

class Repo:
    table = ""
    def check(self, item): ...

    def save(self, item):     # 공통 흐름은 한 곳
        log("saving...")
        self.check(item)
        db.insert(self.table, item)
        log("done")

class UserRepo(Repo):
    table = "users"
    def check(self, u): check_email(u)

class OrderRepo(Repo):
    table = "orders"
    def check(self, o): check_qty(o)
문제

공통 로직을 바꾸려면 복사본 전부를 고쳐야 한다. 한 곳을 놓치면 동작이 엇갈린다.

개선

공통 흐름은 한 곳에서 관리하고, 다른 부분만 오버라이드한다.

문제 05 · 교체·테스트 불가

“테스트 돌릴 때마다 진짜 메일이 나가요”

✕ 절차 지향의존성이 함수 안에 하드코딩

import smtplib

def signup(email):
    save_to_db(email)
    smtp = smtplib.SMTP("mail.server.com")
    smtp.sendmail("me", email, "환영!")


# 테스트하면 실제 메일 발송
# 메일 서비스를 바꾸면 이 함수를 수정
✓ 객체 지향의존성 주입 — 밖에서 넣어준다

class SignupService:
    def __init__(self, mailer):   # 주입받음
        self.mailer = mailer

    def signup(self, email):
        save_to_db(email)
        self.mailer.send(email, "환영!")


# 운영
SignupService(SmtpMailer())
# 테스트 — 가짜 객체로 교체
SignupService(FakeMailer())
# 다른 서비스로 교체
SignupService(SendGridMailer())

외부 시스템과 강하게 결합. 단위 테스트가 어렵고 교체하려면 내부를 고쳐야 한다.

개선

느슨한 결합. 부품을 갈아 끼우듯 운영·테스트·교체가 자유롭다.

Part 3 · AI Agent 시대의 OOP

AI Agent 코딩에서 OOP가 빛나는 이유

AI는 프로젝트 ‘전체’를 한 번에 보지 못한다. 작고 명확한 단위를 줄수록 정확해진다.

01

컨텍스트 절약

AI가 한 번에 읽을 수 있는 양(컨텍스트)은 한정적이다. 잘 나뉜 클래스는 필요한 파일만 보여주면 된다.

예) notifier.py 하나만 첨부 → 정확한 수정
02

인터페이스 = 명세

메서드 이름과 타입 힌트가 곧 프롬프트. “Notifier를 구현해줘” 한 줄이면 충분하다.

예) def send(self, user, msg) 시그니처가 곧 요구사항
03

변경 범위 국소화

새 기능 = 새 클래스. 기존 코드를 덜 건드리니 AI가 망가뜨릴 곳도 줄고, 여러 에이전트가 병렬로 작업해도 충돌이 적다.

예) 새 기능 = 새 파일 1개, 기존 파일 diff 0줄
04

검증이 쉽다

의존성 주입 덕분에 가짜 객체로 즉시 테스트. AI가 쓴 코드를 테스트가 지켜준다.

예) FakeDB로 외부 연결 없이 수 초 만에 검증
→ 사람이 읽기 좋은 구조 = AI가 다루기 좋은 구조

Part 3 · AI Agent 시대의 OOP

AI Agent 자체도 객체로 설계된다

대부분의 Agent 프레임워크는 Tool · Memory · Agent를 클래스로 추상화한다.


class Tool(ABC):
    name: str
    @abstractmethod
    def run(self, query: str) -> str: ...

class WebSearch(Tool):
    name = "web_search"
    def run(self, query): return search_api(query)

class Calculator(Tool):
    name = "calculator"
    def run(self, query): return str(safe_eval(query))

class Agent:
    def __init__(self, tools: list[Tool]):
        self.tools = {t.name: t for t in tools}

    def use(self, name: str, query: str) -> str:
        return self.tools[name].run(query)
1

추상화 — Agent는 run()만 알면 된다. 도구 내부는 몰라도 OK.

2

확장 — 새 도구 = Tool을 상속한 클래스 하나. Agent 코드는 그대로.

3

다형성 — 검색이든 계산이든 같은 방식으로 호출한다.

4

테스트 — 도구마다 독립적으로 검증하고, 가짜 Tool로 Agent를 시험한다.

S

단일 책임 원칙Single Responsibility

한 클래스는 하나의 책임만 가진다. 즉, 변경되는 이유가 하나뿐이어야 한다.※ class만의 이야기가 아니다 — 함수, 그리고 모듈(*.py 파일)도 바뀌는 이유가 하나가 되도록 나눈다. utils.py 하나에 모든 걸 몰아넣지 말 것.

✕ 위반계산 + 표현 + 저장을 한 클래스가

class Report:
    def __init__(self, data):
        self.data = data

    def total(self):            # 계산
        return sum(self.data)

    def to_html(self):          # 표현
        return f"<h1>{self.total()}</h1>"

    def save(self, path):       # 저장
        with open(path, "w") as f:
            f.write(self.to_html())
✓ 준수책임마다 클래스 하나

class Report:                   # 계산만
    def __init__(self, data): self.data = data
    def total(self): return sum(self.data)

class HtmlFormatter:            # 표현만
    def format(self, report):
        return f"<h1>{report.total()}</h1>"

class FileSaver:                # 저장만
    def save(self, text, path):
        with open(path, "w") as f:
            f.write(text)
어기면

HTML 디자인을 고치다 계산 로직이 깨진다. 바뀌는 이유가 3개라 충돌과 회귀 버그가 잦다.

지키면

각자 독립적으로 변경·테스트·재사용. PdfFormatter 추가도 Report를 건드리지 않고 가능.

O

개방-폐쇄 원칙Open / Closed

확장에는 열려 있고, 수정에는 닫혀 있어야 한다. 새 기능은 ‘추가’로 만든다.

✕ 위반등급이 늘 때마다 if 추가

class Checkout:
    def pay(self, price, grade):
        if grade == "vip":
            return price * 0.8
        elif grade == "gold":
            return price * 0.9
        return price


# 새 등급 "platinum"?
# → Checkout 코드를 직접 수정해야 함
✓ 준수Strategy 패턴 — 할인 정책을 객체로

class Discount(ABC):
    @abstractmethod
    def apply(self, price): ...

class Vip(Discount):
    def apply(self, price): return price * 0.8

class Gold(Discount):
    def apply(self, price): return price * 0.9

class Checkout:
    def pay(self, price, discount: Discount):
        return discount.apply(price)

# 새 등급 = 클래스 추가, Checkout은 그대로
class Platinum(Discount):
    def apply(self, price): return price * 0.7
어기면

기능 하나 추가에 잘 돌던 코드를 열어 고친다. 매번 기존 기능 전체를 다시 테스트해야 한다.

지키면

새 정책은 새 클래스로만 추가. 검증된 기존 코드는 안전하게 유지된다.

L

리스코프 치환 원칙Liskov Substitution

부모 타입 자리에 자식을 넣어도 프로그램이 올바르게 동작해야 한다.

✕ 위반자식이 부모의 약속(fly)을 깬다

class Bird:
    def fly(self): return "날아간다"

class Sparrow(Bird): ...

class Penguin(Bird):
    def fly(self): raise Exception("못 날아요!")


def fly_all(birds: list[Bird]):
    for b in birds:
        b.fly()       # 펭귄에서 터짐
✓ 준수모든 새가 지킬 수 있는 약속으로

class Bird:
    def move(self): ...         # 모든 새가 가능

class Sparrow(Bird):
    def move(self): return "날아간다"

class Penguin(Bird):
    def move(self): return "뒤뚱뒤뚱 걷는다"


def move_all(birds: list[Bird]):
    for b in birds:
        b.move()      # 어떤 새든 안전
어기면

부모 타입을 믿을 수 없어 if isinstance(b, Penguin) 같은 예외 처리가 곳곳에 퍼진다.

지키면

다형성을 안심하고 쓸 수 있다. 새 자식 클래스가 와도 호출 코드는 그대로.

I

인터페이스 분리 원칙Interface Segregation

하나의 거대한 인터페이스보다, 역할별로 작게 나눈 여러 인터페이스가 낫다.

✕ 위반모든 기계가 모든 기능을 강요받음

class Machine(ABC):
    @abstractmethod
    def print(self, doc): ...
    @abstractmethod
    def scan(self, doc): ...
    @abstractmethod
    def fax(self, doc): ...

class SimplePrinter(Machine):
    def print(self, doc): ...
    def scan(self, doc):
        raise NotImplementedError   # 억지 구현
    def fax(self, doc):
        raise NotImplementedError   # 억지 구현
✓ 준수역할별 작은 Protocol

class Printer(Protocol):
    def print(self, doc): ...

class Scanner(Protocol):
    def scan(self, doc): ...

class SimplePrinter:          # 필요한 것만
    def print(self, doc): ...

class OfficeMachine:          # 둘 다 가능
    def print(self, doc): ...
    def scan(self, doc): ...

def print_all(p: Printer, docs): ...
어기면

쓰지도 않는 메서드를 억지로 구현하고, 호출하면 런타임에 터진다. 인터페이스 변경 시 모두가 영향받는다.

지키면

각 클래스는 필요한 역할만 구현. 함수는 필요한 능력(Printer)만 요구한다.

D

의존성 역전 원칙Dependency Inversion

고수준 모듈(비즈니스 로직)은 저수준 모듈(DB 등)이 아닌 추상에 의존해야 한다.

✕ 위반OrderService → MySQL (직접 의존)

class MySQL:
    def save(self, data): ...

class OrderService:
    def __init__(self):
        self.db = MySQL()     # 구체 클래스를 직접 생성

    def place(self, order):
        self.db.save(order)


# DB 교체? 테스트?
# → OrderService를 수정해야 함
✓ 준수의존성 주입 — OrderService → Database ← MySQL

class Database(ABC):                  # 추상
    @abstractmethod
    def save(self, data): ...

class MySQL(Database):
    def save(self, data): ...

class FakeDB(Database):               # 테스트용
    def __init__(self): self.saved = []
    def save(self, data): self.saved.append(data)

class OrderService:
    def __init__(self, db: Database):  # 추상에 의존
        self.db = db
    def place(self, order): self.db.save(order)

OrderService(MySQL())    # 운영
OrderService(FakeDB())   # 테스트
어기면

핵심 로직이 DB 같은 세부 구현에 묶인다. 교체·테스트 때마다 핵심 로직을 고쳐야 한다.

지키면

DB, 메일, LLM API까지 부품처럼 교체. 테스트는 FakeDB로 빠르고 안전하게.

Part 5 · 정리

균형 감각: 과설계도 문제다

01

필요할 때, 필요한 만큼

한 번 쓰는 계산에 클래스·추상화를 쌓으면 오히려 읽기 어렵다. YAGNI — 지금 필요 없으면 만들지 말자.


# 과설계
class AbstractAdder(ABC): ...
class AdderFactory: ...
AdderFactory.create("basic").add(1, 2)

# 충분
def add(a, b): return a + b

짧은 스크립트·데이터 변환은 함수/함수형이 더 잘 맞는다. 같은 변경이 반복될 때 추상화하라.

02

상속보다 조합

상속은 강한 결합이다. ‘A는 B이다(is-a)’가 아니라면 ‘A는 B를 가진다(has-a)’로 조합하자.


# 상속: 로봇 오리는 엔진인가?
class RobotDuck(Duck, Engine): ...

# 조합: 로봇 오리는 엔진을 가진다
class RobotDuck(Duck):
    def __init__(self, engine):
        self.engine = engine

상속 계층은 얕게, 기능은 객체를 끼워 넣는 방식으로.

→ 지금은 단순하게, 바뀌는 곳은 유연하게. 원칙은 목적이 아니라 도구다.

Appendix · 생성 패턴

Factory

의도

어떤 클래스를 만들지 결정하는 일을 한 곳에 모은다.

언제 쓰나

  • 문자열·설정값으로 객체를 골라 만들 때
  • 생성 로직이 여기저기 흩어질 때

관련 원칙

OCPDIP

class Dog:
    def speak(self): return "멍멍"

class Cat:
    def speak(self): return "야옹"

class AnimalFactory:
    _registry = {"dog": Dog, "cat": Cat}

    @classmethod
    def create(cls, kind: str):
        return cls._registry[kind]()


pet = AnimalFactory.create("cat")   # 설정값으로 생성
pet.speak()                          # '야옹'
Appendix · 생성 패턴

Singleton

의도

클래스의 인스턴스가 오직 하나만 존재하도록 보장한다.

언제 쓰나

  • 설정, 커넥션 풀처럼 하나만 있어야 할 때

주의

사실상 전역 변수라 남용하면 테스트가 어려워진다. 파이썬에선 모듈 자체가 싱글턴이므로 모듈 변수로 충분한 경우가 많다.


class Config:
    _instance = None

    def __new__(cls):
        if cls._instance is None:
            cls._instance = super().__new__(cls)
        return cls._instance


a = Config()
b = Config()
a is b        # True — 같은 객체
Appendix · 구조 패턴

Adapter

의도

호환되지 않는 인터페이스를 원하는 모양으로 바꿔 연결한다. (돼지코 어댑터)

언제 쓰나

  • 고칠 수 없는 레거시·외부 라이브러리를 쓸 때
  • 여러 LLM API를 같은 인터페이스로 묶을 때

관련 원칙

ISPOCP

class OldPrinter:                  # 바꿀 수 없는 기존 코드
    def print_text(self, s): print(s)

class Printer(Protocol):           # 우리가 원하는 인터페이스
    def output(self, s: str): ...

class PrinterAdapter:
    def __init__(self, old: OldPrinter):
        self.old = old

    def output(self, s):
        self.old.print_text(s)     # 변환해서 전달


def run(p: Printer):
    p.output("hello")

run(PrinterAdapter(OldPrinter()))
Appendix · 구조 패턴

Decorator

의도

객체를 감싸서, 원본을 고치지 않고 기능을 덧붙인다.

언제 쓰나

  • 옵션 조합이 많아 상속으로는 클래스가 폭발할 때
  • 로깅·캐싱·재시도 같은 부가 기능

참고

파이썬의 @decorator 문법은 같은 아이디어를 함수에 적용한 것.


class Coffee:
    def cost(self): return 3000

class Milk:
    def __init__(self, base): self.base = base
    def cost(self): return self.base.cost() + 500

class Shot:
    def __init__(self, base): self.base = base
    def cost(self): return self.base.cost() + 700


# 감싸는 순서대로 기능이 쌓인다
order = Shot(Milk(Coffee()))
order.cost()      # 4200
Appendix · 구조 패턴

Facade

의도

복잡한 하위 시스템 앞에 단순한 창구(메서드) 하나를 둔다.

언제 쓰나

  • 여러 객체를 정해진 순서로 호출해야 할 때
  • 사용하는 쪽에 내부 구조를 숨기고 싶을 때

SRP와의 관계

책임을 잘게 나누면 클래스가 많아진다. Facade는 조율만 맡아, 나눈 부품을 다시 쓰기 쉽게 묶어 준다.


class Inventory:
    def reserve(self, item): ...

class Payment:
    def charge(self, user, price): ...

class Shipping:
    def send(self, item, user): ...

class OrderFacade:                # 단일 창구
    def __init__(self):
        self.inventory = Inventory()
        self.payment = Payment()
        self.shipping = Shipping()

    def place_order(self, user, item, price):
        self.inventory.reserve(item)
        self.payment.charge(user, price)
        self.shipping.send(item, user)


# 호출하는 쪽은 내부를 몰라도 된다
OrderFacade().place_order("kim", "book", 15000)
Appendix · 행위 패턴

Strategy

의도

알고리즘을 객체(또는 함수)로 분리해 실행 중에 갈아 끼운다.

언제 쓰나

  • 같은 일을 하는 방법이 여러 가지일 때
  • if/elif로 방법을 고르고 있을 때

class Catalog:
    def __init__(self, items, sort_key):
        self.items = items
        self.sort_key = sort_key        # 전략 주입

    def show(self):
        return sorted(self.items, key=self.sort_key)

class Channel:
    def __init__(self):
        self.subscribers = []

    def subscribe(self, fn):
        self.subscribers.append(fn)

    def upload(self, title):
        for notify in self.subscribers:
            notify(title)


ch = Channel()
ch.subscribe(lambda t: print("앱 알림:", t))
ch.subscribe(lambda t: print("메일 발송:", t))
ch.upload("OOP 강의")    # 두 구독자 모두 반응

class Pipeline(ABC):
    def run(self):                    # 흐름은 고정
        data = self.load()
        data = self.transform(data)
        self.save(data)

    @abstractmethod
    def load(self): ...
    def transform(self, data): return data   # 기본값
    @abstractmethod
    def save(self, data): ...


class CsvToDb(Pipeline):
    def load(self): return read_csv("a.csv")
    def save(self, data): db.insert(data)
